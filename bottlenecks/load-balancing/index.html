    Balanceamento de carga

    Responsável: Leonardo

    Problema

    Com um único gateway, toda a plataforma depende de um processo: se ele cair, nada responde; se ele saturar, não há como crescer. Como o gateway é a única porta de entrada, ele é o primeiro gargalo.

    Solução

    Um Nginx (lb) passa a ser o único serviço exposto e distribui as requisições entre várias réplicas do gateway (Round Robin, o padrão do bloco upstream). O gateway deixa de publicar porta.

    No EKS há dois níveis: o ELB da AWS espalha o tráfego entre os pods do Nginx (que também têm 2 réplicas, para o próprio balanceador não ser ponto único de falha) e o Service gateway espalha entre os pods do gateway.

    # Bottleneck: balanceamento de carga.
# O Nginx é o único serviço exposto (porta 8080 do host). Ele distribui as
# requisições entre as réplicas do gateway (round robin, o padrão do upstream).

user  nginx;
worker_processes  auto;

error_log  /var/log/nginx/error.log notice;
pid        /run/nginx.pid;

events {
    worker_connections  1024;
}

http {

    # DNS interno do Docker: com "resolve" (nginx >= 1.27.3) o upstream é
    # re-resolvido a cada 10 s, então "docker compose up --scale gateway=N"
    # entra no balanceamento sem precisar de nginx -s reload
    resolver 127.0.0.11 valid=10s ipv6=off;

    upstream gateways {
        zone gateways 64k;
        server gateway:8080 resolve;
        # conexões reaproveitadas com o gateway (menos handshakes sob carga)
        keepalive 32;
    }

    log_format upstream '$remote_addr "$request" $status -> $upstream_addr ($upstream_response_time s)';
    access_log /var/log/nginx/access.log upstream;

    server {
        listen 80;
        server_tokens off;

        # métricas e detalhes do actuator ficam só na rede interna
        # (o Prometheus fala direto com o gateway); o health fica público
        location = /gateway/actuator/health {
            proxy_pass http://gateways;
        }
        location /gateway/actuator {
            return 404;
        }

        location / {
            proxy_pass         http://gateways;
            proxy_http_version 1.1;
            proxy_set_header   Connection        "";
            proxy_set_header   Host              $host;
            proxy_set_header   X-Real-IP         $remote_addr;
            proxy_set_header   X-Forwarded-For   $proxy_add_x_forwarded_for;
            proxy_set_header   X-Forwarded-Proto $scheme;
        }
    }
}

    Pontos além do hands-on 7 do curso:

    Diretiva Por quê
    resolver 127.0.0.11 + server gateway:8080 resolve O Nginx re-resolve o DNS do Docker a cada 10 s. Com docker compose up --scale gateway=N as réplicas novas entram no balanceamento sem nginx -s reload (no hands-on o Nginx só resolve na inicialização).
    keepalive 32 + proxy_http_version 1.1 Reaproveita conexões com o gateway: menos handshakes TCP sob carga.
    keepalive_time 10s (só no k8s/lb.yaml) No k8s o Nginx fala com o Service gateway, que balanceia conexões, não requisições: uma conexão aberta fica no mesmo pod. Fechando cada conexão depois de 10 s, as réplicas que o HPA cria entram na divisão em ≈ 35–40 s em vez de ≈ 1,5 min (achado do teste com HPA). No compose não precisa: lá o Nginx enxerga cada réplica pelo DNS e divide as requisições ele mesmo.
    location /gateway/actuator { return 404; } Métricas do actuator ficam só na rede interna (o Prometheus fala direto com o gateway).
    log_format upstream O access log mostra qual réplica ($upstream_addr) atendeu cada requisição.

    Evidência

    O GET /health-check do gateway responde com o hostname do container. Com 3 réplicas, 15 requisições pelo lb se dividem igualmente — e a terceira réplica entrou com o compose já rodando, sem reload do Nginx:

    $ docker compose up -d --scale gateway=3 --no-recreate
$ for i in $(seq 1 15); do curl -s localhost:8080/health-check; echo; done | sort | uniq -c
   5 {"hostname":"112e1c20be21"}
   5 {"hostname":"5f9dceeefb1f"}
   5 {"hostname":"c98aacd33a1e"}

    O mesmo teste roda no CI (scripts/smoke-test.sh, workflow integration), que falha se menos de duas réplicas responderem.

    # Bottleneck: balanceamento de carga no EKS.
#
#   internet -> ELB da AWS (Service type LoadBalancer)
#            -> pods do Nginx (lb, 2 réplicas)
#            -> Service gateway (ClusterIP) -> pods do gateway (2+ réplicas)
#
# O Nginx é a única porta de entrada: gateway, auth, account e exchange são
# ClusterIP (camada confiável), como no compose.
apiVersion: v1
kind: ConfigMap
metadata:
  name: lb-configmap
data:
  nginx.conf: |
    worker_processes auto;
    pid /tmp/nginx.pid;
    error_log /dev/stderr notice;

    events {
        worker_connections 1024;
    }

    http {
        client_body_temp_path /tmp/client_temp;
        proxy_temp_path       /tmp/proxy_temp;
        fastcgi_temp_path     /tmp/fastcgi_temp;
        uwsgi_temp_path       /tmp/uwsgi_temp;
        scgi_temp_path        /tmp/scgi_temp;

        log_format upstream '$remote_addr "$request" $status -> $upstream_addr ($upstream_response_time s)';
        access_log /dev/stdout upstream;

        # no k8s o Service "gateway" já é um IP virtual que distribui entre
        # os pods; o Nginx mantém conexões abertas com ele (keepalive)
        upstream gateways {
            server gateway:8080;
            keepalive 32;
            # o Service balanceia conexões, não requisições: sem limite, cada
            # conexão fica no mesmo pod por 1000 requisições e as réplicas que o
            # HPA cria levam ~1,5 min para receber carga (~35-40 s com 10 s)
            keepalive_time 10s;
        }

        server {
            listen 8080;
            server_tokens off;

            location = /lb/health {
                access_log off;
                return 200 "ok\n";
            }

            location = /gateway/actuator/health {
                proxy_pass http://gateways;
            }
            location /gateway/actuator {
                return 404;
            }

            location / {
                proxy_pass         http://gateways;
                proxy_http_version 1.1;
                proxy_set_header   Connection        "";
                proxy_set_header   Host              $host;
                proxy_set_header   X-Real-IP         $remote_addr;
                proxy_set_header   X-Forwarded-For   $proxy_add_x_forwarded_for;
                proxy_set_header   X-Forwarded-Proto $scheme;
            }
        }
    }
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: lb
  labels:
    app: lb
spec:
  replicas: 2
  selector:
    matchLabels:
      app: lb
  template:
    metadata:
      labels:
        app: lb
    spec:
      securityContext:
        runAsNonRoot: true
        runAsUser: 101
        runAsGroup: 101
        seccompProfile:
          type: RuntimeDefault
      containers:
        - name: nginx
          image: nginx:1.29-alpine
          ports:
            - containerPort: 8080
          readinessProbe:
            httpGet:
              path: /lb/health
              port: 8080
            periodSeconds: 5
          resources:
            requests:
              memory: "32Mi"
              cpu: "50m"
            limits:
              memory: "128Mi"
              cpu: "500m"
          securityContext:
            allowPrivilegeEscalation: false
            readOnlyRootFilesystem: true
            capabilities:
              drop: [ "ALL" ]
          volumeMounts:
            - name: config
              mountPath: /etc/nginx/nginx.conf
              subPath: nginx.conf
              readOnly: true
            - name: tmp
              mountPath: /tmp
      volumes:
        - name: config
          configMap:
            name: lb-configmap
        - name: tmp
          emptyDir: {}
---
apiVersion: v1
kind: Service
metadata:
  name: lb
spec:
  type: LoadBalancer
  selector:
    app: lb
  ports:
    - name: http
      protocol: TCP
      port: 80
      targetPort: 8080
