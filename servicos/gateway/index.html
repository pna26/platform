    Gateway

    Responsável: Leonardo · Repositório: pna26/gateway-service

    O gateway é o ponto único de entrada da plataforma. Ele:

    1. roteia cada caminho para o microsserviço certo (/auth/**, /accounts/**, /exchange/**);
    2. autoriza: em toda rota que não é aberta, exige o cookie __store_jwt_token, valida o token no Auth (POST /auth/solve) e repassa o id da conta no header id-account;
    3. descarta qualquer id-account enviado pelo cliente, para ninguém se passar por outra conta;
    4. guarda no Redis o resultado da validação do token (bottleneck de cache);
    5. expõe GET /health-check com o hostname da réplica (mostra o balanceamento).

    É implementado com Spring Cloud Gateway sobre WebFlux (reativo), como no hands-on 2 e 3 do curso.

    Rotas

    Caminho Destino Token?
    POST /auth/register, POST /auth/login, GET /auth/logout auth:8080 não
    demais /auth/** auth:8080 sim
    /accounts/** account:8080 sim
    /exchange/** exchange:8080 sim
    /exchanges/** exchange:8080, reescrito para /exchange/** sim
    GET /health-check o próprio gateway não
    /gateway/actuator/health, /gateway/actuator/prometheus o próprio gateway não (o lb bloqueia o prometheus para fora)

    Os destinos vêm de variáveis de ambiente (AUTH_URI, ACCOUNT_URI, EXCHANGE_URI), com os nomes dos serviços do compose/k8s como padrão.

    Decisão de autorização

    flowchart TD
    A[Requisição] --> B{rota aberta?<br/>RouterValidator}
    B -- sim --> OK[encaminha ao serviço]
    B -- não --> C{cookie<br/>__store_jwt_token?}
    C -- não --> E401[401]
    C -- sim --> D{token no Redis?}
    D -- sim --> H[header id-account = id em cache]
    D -- não --> F[POST /auth/solve]
    F -- 4xx --> E401
    F -- erro/timeout --> E503[503]
    F -- 200 --> G[grava no Redis<br/>TTL = min 5 min, exp]
    G --> H
    H --> OK

    package store.gateway.security;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.cloud.gateway.filter.GatewayFilterChain;
import org.springframework.cloud.gateway.filter.GlobalFilter;
import org.springframework.core.Ordered;
import org.springframework.http.HttpCookie;
import org.springframework.http.HttpStatus;
import org.springframework.http.server.reactive.ServerHttpRequest;
import org.springframework.stereotype.Component;
import org.springframework.web.server.ServerWebExchange;

import reactor.core.publisher.Mono;

/**
 * Filtro global de autorização.
 *
 * 1. rota aberta (RouterValidator)  -> segue sem token;
 * 2. sem cookie __store_jwt_token   -> 401;
 * 3. token no cache (Redis)         -> usa o id em cache;
 * 4. senão, POST /auth/solve        -> 401 se inválido, 503 se o Auth estiver fora;
 * 5. repassa o id no header id-account para o serviço de destino.
 *
 * O header id-account enviado pelo cliente já foi removido pelo default-filter
 * RemoveRequestHeader (application.yaml), que roda antes deste filtro.
 */
@Component
public class AuthorizationFilter implements GlobalFilter, Ordered {

    private static final Logger logger = LoggerFactory.getLogger(AuthorizationFilter.class);

    public static final String AUTH_COOKIE_TOKEN = "__store_jwt_token";
    public static final String HEADER_ID_ACCOUNT = "id-account";

    // depois dos default-filters (ordens 1, 2, ...) e antes do roteamento
    public static final int ORDER = 100;

    private final RouterValidator routerValidator;
    private final AuthClient authClient;
    private final TokenCache tokenCache;

    public AuthorizationFilter(RouterValidator routerValidator, AuthClient authClient, TokenCache tokenCache) {
        this.routerValidator = routerValidator;
        this.authClient = authClient;
        this.tokenCache = tokenCache;
    }

    @Override
    public int getOrder() {
        return ORDER;
    }

    @Override
    public Mono<Void> filter(ServerWebExchange exchange, GatewayFilterChain chain) {
        final ServerHttpRequest request = exchange.getRequest();

        if (!routerValidator.isSecured.test(request)) {
            logger.debug("rota aberta: {} {}", request.getMethod(), request.getURI().getPath());
            return chain.filter(exchange);
        }

        final HttpCookie cookie = request.getCookies().getFirst(AUTH_COOKIE_TOKEN);
        final String jwt = cookie == null ? null : cookie.getValue().trim();
        if (jwt == null || jwt.isEmpty()) {
            logger.debug("sem cookie: {} {}", request.getMethod(), request.getURI().getPath());
            return reject(exchange, HttpStatus.UNAUTHORIZED);
        }

        // os erros tratados aqui são só os da validação do token; erros do
        // serviço de destino (depois do chain.filter) seguem o fluxo normal
        return resolve(jwt)
            .flatMap(result -> result.idAccount() != null
                ? chain.filter(withIdAccount(exchange, result.idAccount()))
                : reject(exchange, result.status()));
    }

    private Mono<Resolution> resolve(String jwt) {
        return tokenCache.get(jwt)
            .switchIfEmpty(Mono.defer(() -> authClient.solve(jwt)
                .flatMap(id -> tokenCache.put(jwt, id).thenReturn(id))))
            .map(Resolution::ok)
            .onErrorResume(AuthClient.InvalidTokenException.class,
                e -> Mono.just(Resolution.fail(HttpStatus.UNAUTHORIZED)))
            .onErrorResume(e -> {
                logger.error("auth indisponível: {}", e.toString());
                return Mono.just(Resolution.fail(HttpStatus.SERVICE_UNAVAILABLE));
            });
    }

    private record Resolution(String idAccount, HttpStatus status) {
        static Resolution ok(String idAccount) { return new Resolution(idAccount, null); }
        static Resolution fail(HttpStatus status) { return new Resolution(null, status); }
    }

    private static ServerWebExchange withIdAccount(ServerWebExchange exchange, String idAccount) {
        return exchange.mutate()
            .request(exchange.getRequest().mutate()
                .headers(h -> h.set(HEADER_ID_ACCOUNT, idAccount))
                .build())
            .build();
    }

    private static Mono<Void> reject(ServerWebExchange exchange, HttpStatus status) {
        if (exchange.getResponse().isCommitted()) return Mono.empty();
        exchange.getResponse().setStatusCode(status);
        return exchange.getResponse().setComplete();
    }

}

    package store.gateway.security;

import java.util.List;
import java.util.function.Predicate;

import org.springframework.http.server.reactive.ServerHttpRequest;
import org.springframework.stereotype.Component;

/**
 * Lista das rotas abertas (que não exigem o cookie JWT).
 * Cada entrada é "MÉTODO /caminho"; o método ANY casa com qualquer verbo e o
 * sufixo /** casa com qualquer subcaminho.
 */
@Component
public class RouterValidator {

    public static final List<String> OPEN_API_ENDPOINTS = List.of(
        "POST /auth/register",
        "POST /auth/login",
        "GET /auth/logout"
    );

    private final List<Endpoint> openEndpoints = OPEN_API_ENDPOINTS.stream()
        .map(Endpoint::parse)
        .toList();

    public final Predicate<ServerHttpRequest> isSecured = request -> openEndpoints
        .stream()
        .noneMatch(e -> e.matches(request.getMethod().name(), request.getURI().getPath()));

    record Endpoint(String method, String path, boolean deep) {

        static Endpoint parse(String spec) {
            final String[] parts = spec.trim().split("\\s+");
            final String path = parts[1];
            final boolean deep = path.endsWith("/**");
            return new Endpoint(parts[0], deep ? path.substring(0, path.length() - 3) : path, deep);
        }

        boolean matches(String requestMethod, String requestPath) {
            final boolean methodOk = "ANY".equalsIgnoreCase(method) || method.equalsIgnoreCase(requestMethod);
            if (!methodOk) return false;
            // "/auth/login/" é tratado igual a "/auth/login"
            final String normalized = requestPath.length() > 1 && requestPath.endsWith("/")
                ? requestPath.substring(0, requestPath.length() - 1)
                : requestPath;
            return deep
                ? normalized.equals(path) || normalized.startsWith(path + "/")
                : normalized.equals(path);
        }
    }

}

    package store.gateway.security;

import java.time.Duration;
import java.util.Map;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.core.ParameterizedTypeReference;
import org.springframework.http.HttpStatusCode;
import org.springframework.http.MediaType;
import org.springframework.stereotype.Component;
import org.springframework.web.reactive.function.client.WebClient;

import reactor.core.publisher.Mono;

/**
 * Cliente do Auth: troca o JWT pelo id da conta chamando POST /auth/solve.
 */
@Component
public class AuthClient {

    private static final ParameterizedTypeReference<Map<String, String>> MAP_TYPE =
        new ParameterizedTypeReference<>() {};

    private final WebClient webClient;
    private final Duration timeout;

    public AuthClient(
        WebClient.Builder builder,
        @Value("${store.gateway.auth.uri:http://auth:8080}") String authUri,
        @Value("${store.gateway.auth.timeout:3s}") Duration timeout
    ) {
        this.webClient = builder.baseUrl(authUri).build();
        this.timeout = timeout;
    }

    /**
     * @return o id da conta dono do token; {@link InvalidTokenException} se o
     *         Auth recusar o token (4xx); outro erro se o Auth estiver fora.
     */
    public Mono<String> solve(String jwt) {
        return webClient.post()
            .uri("/auth/solve")
            .contentType(MediaType.APPLICATION_JSON)
            .bodyValue(Map.of("token", jwt))
            .retrieve()
            // o auth responde 401 para token inválido/expirado: continua 401 no gateway
            .onStatus(HttpStatusCode::is4xxClientError,
                response -> Mono.error(new InvalidTokenException()))
            .bodyToMono(MAP_TYPE)
            .timeout(timeout)
            .flatMap(body -> {
                final String id = body.get("idAccount");
                return id == null || id.isBlank()
                    ? Mono.error(new InvalidTokenException())
                    : Mono.just(id);
            });
    }

    public static class InvalidTokenException extends RuntimeException {
        public InvalidTokenException() {
            super("Invalid token", null, false, false);
        }
    }

}

    package store.gateway.security;

import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.time.Duration;
import java.time.Instant;
import java.util.Base64;
import java.util.HexFormat;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.data.redis.core.ReactiveStringRedisTemplate;
import org.springframework.stereotype.Component;

import io.micrometer.core.instrument.Counter;
import io.micrometer.core.instrument.MeterRegistry;
import reactor.core.publisher.Mono;

/**
 * Bottleneck: cache (Redis).
 *
 * Sem cache, TODA requisição autenticada faz uma chamada HTTP ao Auth
 * (POST /auth/solve) para validar o JWT. Aqui guardamos no Redis o resultado
 * (hash do token -> id da conta) por alguns minutos, compartilhado entre todas
 * as réplicas do gateway. Assim o Auth só é chamado no primeiro acesso de cada
 * token (cache miss).
 *
 * Decisões:
 * - a chave é o SHA-256 do token, nunca o token em si;
 * - o TTL nunca passa do "exp" do JWT (token expirado não fica no cache);
 * - se o Redis cair, o gateway continua funcionando chamando o Auth (fail-open
 *   só do cache, a validação continua obrigatória).
 */
@Component
public class TokenCache {

    private static final Logger logger = LoggerFactory.getLogger(TokenCache.class);
    private static final String PREFIX = "gateway:token:";
    private static final Pattern EXP = Pattern.compile("\"exp\"\\s*:\\s*(\\d+)");

    private final ReactiveStringRedisTemplate redis;
    private final boolean enabled;
    private final Duration ttl;
    private final Counter hits;
    private final Counter misses;
    private final Counter errors;

    public TokenCache(
        ReactiveStringRedisTemplate redis,
        MeterRegistry registry,
        @Value("${store.gateway.cache.enabled:true}") boolean enabled,
        @Value("${store.gateway.cache.ttl:5m}") Duration ttl
    ) {
        this.redis = redis;
        this.enabled = enabled;
        this.ttl = ttl;
        this.hits = counter(registry, "hit");
        this.misses = counter(registry, "miss");
        this.errors = counter(registry, "error");
    }

    private static Counter counter(MeterRegistry registry, String result) {
        // hit = resolvido pelo Redis; miss = precisou chamar o Auth;
        // error = Redis fora (o gateway segue sem cache)
        return Counter.builder("gateway.token.cache")
            .description("Resolucao do token pelo cache Redis, por resultado (hit, miss, error)")
            .tag("result", result)
            .register(registry);
    }

    /** id da conta em cache para o token, ou vazio (miss, cache desligado ou Redis fora). */
    public Mono<String> get(String jwt) {
        if (!enabled) return Mono.empty();
        return redis.opsForValue().get(key(jwt))
            .doOnNext(id -> hits.increment())
            .switchIfEmpty(Mono.defer(() -> {
                misses.increment();
                return Mono.empty();
            }))
            .onErrorResume(e -> {
                errors.increment();
                logger.warn("redis indisponível, seguindo sem cache: {}", e.getMessage());
                return Mono.empty();
            });
    }

    /** guarda token -> id com TTL = min(ttl configurado, tempo até o exp do JWT). */
    public Mono<Void> put(String jwt, String idAccount) {
        if (!enabled) return Mono.empty();
        final Duration effective = effectiveTtl(jwt, Instant.now());
        if (effective.isZero() || effective.isNegative()) return Mono.empty();
        return redis.opsForValue().set(key(jwt), idAccount, effective)
            .then()
            .onErrorResume(e -> {
                errors.increment();
                logger.warn("não consegui gravar no redis: {}", e.getMessage());
                return Mono.empty();
            });
    }

    Duration effectiveTtl(String jwt, Instant now) {
        final Long exp = expiration(jwt);
        if (exp == null) return ttl;
        final Duration untilExp = Duration.between(now, Instant.ofEpochSecond(exp));
        return untilExp.compareTo(ttl) < 0 ? untilExp : ttl;
    }

    /** lê o "exp" do payload; o token já foi validado pelo Auth antes de chegar aqui. */
    static Long expiration(String jwt) {
        try {
            final String[] parts = jwt.split("\\.");
            if (parts.length < 2) return null;
            final String payload = new String(Base64.getUrlDecoder().decode(parts[1]), StandardCharsets.UTF_8);
            final Matcher m = EXP.matcher(payload);
            return m.find() ? Long.valueOf(m.group(1)) : null;
        } catch (IllegalArgumentException e) {
            return null;
        }
    }

    static String key(String jwt) {
        try {
            final byte[] digest = MessageDigest.getInstance("SHA-256")
                .digest(jwt.getBytes(StandardCharsets.UTF_8));
            return PREFIX + HexFormat.of().formatHex(digest);
        } catch (NoSuchAlgorithmException e) {
            throw new IllegalStateException(e);
        }
    }

}

        webflux:

              '[/**]':

                allowedMethods: "*"
                allowCredentials: true

          httpclient:
            connect-timeout: 2000
            response-timeout: 10s

          routes:

            - id: auth
              uri: ${AUTH_URI:http://auth:8080}
              predicates:
                - Path=/auth/**

            - id: account
              uri: ${ACCOUNT_URI:http://account:8080}
              predicates:
                - Path=/accounts/**

            - id: exchange
              uri: ${EXCHANGE_URI:http://exchange:8080}
              predicates:
                - Path=/exchange/**

            # o enunciado usa /exchange e /exchanges: o contrato fechou /exchange,
            # e /exchanges continua funcionando como apelido
            - id: exchanges-alias
              uri: ${EXCHANGE_URI:http://exchange:8080}
              predicates:
                - Path=/exchanges/**
              filters:
                - RewritePath=/exchanges/(?<segment>.*), /exchange/$\{segment}

store:
  gateway:
    auth:
      uri: ${AUTH_URI:http://auth:8080}
      timeout: ${AUTH_TIMEOUT:3s}
    cache:
      enabled: ${TOKEN_CACHE_ENABLED:true}
      ttl: ${TOKEN_CACHE_TTL:5m}

management:
  endpoints:
    web:
      base-path: /gateway/actuator

        include: [ 'health', 'info', 'prometheus' ]
  endpoint:
    health:
      probes:
        enabled: true
  health:
    # o redis é só cache: se cair, o gateway continua respondendo (sem cache)

      enabled: false

    GatewayRoutingTest.java (trecho)
        @Test
    void idAccountForjadoPeloClienteEhDescartado() {
        // sem token: o header forjado não abre a rota
        client.get().uri("/accounts").header("id-account", "999")
            .exchange().expectStatus().isUnauthorized();
        // com token: o serviço recebe o id do token, não o forjado
        client.get().uri("/accounts").header("id-account", "999").cookie("__store_jwt_token", "jwt-ok")
            .exchange()
            .expectStatus().isOk()
            .expectBody().jsonPath("$.idAccount").isEqualTo("42");
    }
